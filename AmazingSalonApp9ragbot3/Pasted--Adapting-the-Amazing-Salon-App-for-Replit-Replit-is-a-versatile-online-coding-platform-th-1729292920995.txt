## **1. Appointment Scheduling (Team 1)**
**Languages/Frameworks:**  
- **Frontend:** React.js  
- **Backend:** Node.js with Express.js  
- **Database:** Replit DB (for simplicity) or an external PostgreSQL service like [Supabase](https://supabase.com/)  

### **Setting Up the Backend**

1. **Create a New Repl:**
   - Go to [Replit](https://replit.com/) and create a new Repl using the **Node.js** template.

2. **Install Dependencies:**
   - In the Repl's shell, install necessary packages:
     ```bash
     npm install express cors axios
     ```

3. **Set Up Replit DB (Optional):**
   - For simplicity, we'll use Replit's built-in database. If you prefer PostgreSQL, consider using Supabase and connect via its API.

4. **Create the Server:**
   - **`index.js`**
     ```javascript
     const express = require('express');
     const cors = require('cors');
     const app = express();
     const PORT = process.env.PORT || 3000;

     // Use Replit DB or external database here
     const Database = require("@replit/database");
     const db = new Database();

     app.use(cors());
     app.use(express.json());

     // Routes
     app.post('/api/appointments', async (req, res) => {
       const { stylist, service, date, time } = req.body;
       const appointment = { stylist, service, date, time, id: Date.now() };
       await db.set(`appointment_${appointment.id}`, appointment);
       res.status(201).json(appointment);
     });

     app.get('/api/appointments', async (req, res) => {
       const keys = await db.list();
       const appointments = [];
       for (const key of keys) {
         if (key.startsWith("appointment_")) {
           const appointment = await db.get(key);
           appointments.push(appointment);
         }
       }
       res.json(appointments);
     });

     app.listen(PORT, () => {
       console.log(`Server is running on port ${PORT}`);
     });
     ```

5. **Run the Server:**
   - Click the **Run** button in Replit. The server will start, and Replit will provide a URL to access it.

     ```python
     from flask import Flask, request, jsonify
     from flask_cors import CORS
     import os
     import time

     app = Flask(__name__)
     CORS(app)

     # Simple in-memory inventory using a dictionary
     inventory = {}

     @app.route('/api/products', methods=['POST'])
     def add_product():
         data = request.json
         product_id = int(time.time())
         inventory[product_id] = {
             'id': product_id,
             'name': data['name'],
             'barcode': data['barcode'],
             'quantity': data['quantity'],
             'reorder_threshold': data['reorder_threshold'],
             'supplier': data['supplier']
         }
         return jsonify(inventory[product_id]), 201

     @app.route('/api/products', methods=['GET'])
     def get_products():
         return jsonify(list(inventory.values()))

     if __name__ == '__main__':
         app.run(host='0.0.0.0', port=3000)
     ```


3. **Create the Payment Form Component:**
   - **`src/components/PaymentForm.jsx`**
     ```jsx
     import React, { useState } from 'react';
     import { CardElement, useStripe, useElements } from '@stripe/react-stripe-js';
     import axios from 'axios';

     const PaymentForm = () => {
       const stripe = useStripe();
       const elements = useElements();
       const [amount, setAmount] = useState('');

       const handleSubmit = async (e) => {
         e.preventDefault();
         if (!stripe || !elements) return;

         const card = elements.getElement(CardElement);
         const { error, paymentMethod } = await stripe.createPaymentMethod({
           type: 'card',
           card,
         });

         if (error) {
           console.error(error);
           alert(error.message);
         } else {
           try {
             const response = await axios.post('https://YOUR_BACKEND_URL/api/payments', {
               amount,
               payment_method_id: paymentMethod.id
             });
             if (response.data.success) {
               alert('Payment Successful!');
             } else {
               alert('Payment Failed: ' + response.data.error);
             }
           } catch (err) {
             console.error(err);
             alert('Payment Error');
           }
         }
       };

       return (
         <form onSubmit={handleSubmit}>
           <input
             type="number"
             placeholder="Amount"
             value={amount}
             onChange={(e) => setAmount(e.target.value)}
             required
           />
           <CardElement />
           <button type="submit" disabled={!stripe}>
             Pay
           </button>
         </form>
       );
     };

     export default PaymentForm;
     ```

   - **`src/App.js`**

     ```python
     from flask import Flask, request, jsonify
     from flask_cors import CORS
     import os
     from collections import defaultdict

     app = Flask(__name__)
     CORS(app)

     # Mock data for appointments
     appointments = [
         {'id': 1, 'service': 'Haircut', 'price': 50, 'date': '2024-01-15'},
         {'id': 2, 'service': 'Coloring', 'price': 80, 'date': '2024-01-16'},
         # Add more mock data as needed
     ]

     @app.route('/api/reports/revenue', methods=['GET'])
     def revenue_report():
         start = request.args.get('start')
         end = request.args.get('end')
         total = sum(app['price'] for app in appointments if start <= app['date'] <= end)
         return jsonify({'total': total})

     @app.route('/api/reports/popular-services', methods=['GET'])
     def popular_services():
         service_count = defaultdict(int)
         for app in appointments:
             service_count[app['service']] += 1
         sorted_services = sorted(service_count.items(), key=lambda x: x[1], reverse=True)
         return jsonify(sorted_services)

     if __name__ == '__main__':
         app.run(host='0.0.0.0', port=3000)
     ```

4. **Create Authentication Routes:**
   - **`routes/auth.js`**
     ```javascript
     const express = require('express');
     const router = express.Router();
     const bcrypt = require('bcrypt');
     const jwt = require('jsonwebtoken');
     const db = require('@replit/database')();

     // User Registration (Optional)
     router.post('/register', async (req, res) => {
       const { email, password, role } = req.body;
       try {
         // Check if user exists
         const keys = await db.list();
         const userExists = keys.some(key => key === `user_${email}`);
         if (userExists) return res.status(400).send('User already exists');

         // Hash password
         const hashedPassword = await bcrypt.hash(password, 10);
         const user = { email, password: hashedPassword, role };
         await db.set(`user_${email}`, user);
         res.status(201).send('User registered');
       } catch (error) {
         console.error(error);
         res.status(500).send('Server Error');
       }
     });

     // User Login
     router.post('/login', async (req, res) => {
       const { email, password } = req.body;
       try {
         const user = await db.get(`user_${email}`);
         if (!user) return res.status(400).send('Invalid Credentials');

         const validPassword = await bcrypt.compare(password, user.password);
         if (!validPassword) return res.status(400).send('Invalid Credentials');

         const token = jwt.sign({ email: user.email, role: user.role }, 'your_jwt_secret', { expiresIn: '1h' });
         res.json({ token });
       } catch (error) {
         console.error(error);
         res.status(500).send('Server Error');
       }
     });

     module.exports = router;
     ```

5. **Protect Routes with Authentication Middleware:**
   - **`middleware/auth.js`**
     ```javascript
     const jwt = require('jsonwebtoken');

     const auth = (req, res, next) => {
       const token = req.header('Authorization')?.split(' ')[1];
       if (!token) return res.status(401).send('Access Denied');

       try {
         const verified = jwt.verify(token, 'your_jwt_secret');
         req.user = verified;
         next();
       } catch (err) {
         res.status(400).send('Invalid Token');
       }
     };

     module.exports = auth;
     ```
